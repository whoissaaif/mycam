# Golden byte sequences for the MyCam wire protocol (see PROTOCOL.md).
# Both test suites check against this file:
#   Android: app/src/test/java/io/github/whoissaaif/mycam/ProtocolTest.kt  (builds packets, parses commands)
#   Windows: pc/tests/tests.cpp                                   (parses packets, builds commands)
# Format: name = hex bytes (spaces are ignored). Change both implementations and this file together.

# Phone -> PC packets: magic | type | flags | reserved(2) | ptsUs(8) | length(4) | payload
packet.hello   = 4D43414D 00 00 0000 0000000000000000 00000002 0001
packet.config  = 4D43414D 01 00 0000 0000000000000000 0000000C 0780 0438 005A 01 0000000167
packet.frame   = 4D43414D 02 01 0000 0102030405060708 00000004 DEADBEEF
packet.orient  = 4D43414D 03 00 0000 0000000000000000 00000002 010E
packet.state   = 4D43414D 04 00 0000 0000000000000000 00000002 01 01
packet.state_paused = 4D43414D 04 00 0000 0000000000000000 00000002 03 00
packet.log     = 4D43414D 05 00 0000 0000000000000000 00000002 6869

# PC -> phone commands: magic | cmd | arg | reserved(2)
command.hello       = 4D434D44 01 00 0000
command.start       = 4D434D44 02 00 0000
command.stop        = 4D434D44 03 00 0000
command.keyframe    = 4D434D44 04 00 0000
command.facing_front = 4D434D44 05 01 0000
command.pause       = 4D434D44 06 00 0000
command.resume      = 4D434D44 07 00 0000

# v3: camera settings and capabilities (phone -> PC), and the commands that change them (PC -> phone).
# CAMERA payload: quality | fps | zoom x100 (u16) | zoomMin x100 | zoomMax x100 | ev | evMin | evMax |
#                 evStep x100 | flags | width (u16) | height (u16) | actual fps
# here: 1080p, 30 fps, zoom 1.00 (0.60..10.00), EV 0 (-12..+12, step 0.33), flags torch+has60+has4K,
#       actual 1920x1080 @ 30
packet.camera  = 4D43414D 06 00 0000 0000000000000000 00000012 01 1E 0064 003C 03E8 00 F4 0C 21 19 0780 0438 1E
command.quality_4k  = 4D434D44 08 02 0000
command.fps_60      = 4D434D44 09 3C 0000
command.zoom_2x     = 4D434D44 0A 14 0000
command.exposure_m2 = 4D434D44 0B FE 0000
command.torch_on    = 4D434D44 0C 01 0000
command.focus_lock  = 4D434D44 0D 01 0000
